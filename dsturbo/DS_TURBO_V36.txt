    // 3'lü grup geliş seçeneklerinin tamamında: Hasta Geçmişi'ni kapat,

    async function ucGrupLaboratuvariHazirla() {
        // Hasta Geçmişi bu oturumda daha önce kapatılmadıysa kapat.
        if (!window.fonetHastaGecmisiKapatildi) {
            const kapatmaSimgeleri = [
                ...document.querySelectorAll("img.x-tool-img.x-tool-close")

            const hastaGecmisiKapat = kapatmaSimgeleri[kapatmaSimgeleri.length - 1];
            if (!hastaGecmisiKapat) {
                throw new Error("Hasta Geçmişi penceresinin kapatma simgesi bulunamadı.");
            }

            hastaGecmisiKapat.click();
            window.fonetHastaGecmisiKapatildi = true;
            await bekle(600);
            console.log("Hasta Geçmişi penceresi kapatıldı.");
        } else {
            console.log("Hasta Geçmişi penceresi bu akışta zaten kapatılmış.");
        }

        // Laboratuvar sekmesine geç.
        const labYazi = [...document.querySelectorAll("span.x-tab-inner")]
            .find(el =>
                el.getClientRects().length > 0 &&
                el.textContent.trim() === "Laboratuvar"
            );
        const labSekmesi = labYazi?.closest("a.x-tab");

        if (!labSekmesi) {
            throw new Error("Laboratuvar sekmesi bulunamadı.");
        }

        labSekmesi.click();
        await bekle(700);

        // Doğru açılır kutu, dinamik ID yerine input[name="stokDurumu"] ile belirlenir.
        const stokDurumuAlanlari = [...document.querySelectorAll('input[name="stokDurumu"]')]
            .filter(el => el.getClientRects().length > 0);

        if (stokDurumuAlanlari.length !== 1) {
            throw new Error(
                "Laboratuvar stok durumu alanı tekil olarak bulunamadı. Görünür alan sayısı: " +
                stokDurumuAlanlari.length
            );
        }

        const stokDurumu = stokDurumuAlanlari[0];
        const stokDurumuOku = stokDurumu.closest("tr")
            ?.querySelector(".x-form-arrow-trigger");

        if (!stokDurumuOku) {
            throw new Error("Tüm Gelişler açılır kutusunun oku bulunamadı.");
        }

        stokDurumuOku.click();
        await bekle(400);

        // Açılan listede yalnızca tam metni eşleşen görünür seçeneği seç.
        let tumGelisSecenegi = null;
        const secenekBaslangic = Date.now();

        while (!tumGelisSecenegi && Date.now() - secenekBaslangic < 5000) {
            tumGelisSecenegi = [...document.querySelectorAll(".x-boundlist-item")]
                .find(el =>
                    el.getClientRects().length > 0 &&
                    el.textContent.trim() === "Tüm Gelişler"
                );

            if (!tumGelisSecenegi) await bekle(200);
        }

        if (!tumGelisSecenegi) {
            throw new Error("Açılır listede Tüm Gelişler seçeneği bulunamadı.");
        }

        tumGelisSecenegi.click();
        await bekle(350);

        if (stokDurumu.value.trim() !== "Tüm Gelişler") {
            throw new Error(
                "Tüm Gelişler seçimi kutuda doğrulanamadı. Görünen değer: " +
                (stokDurumu.value || "(boş)")

                window.fonetF8Running = true;

                } finally {

                    butonlariKilitle(false);
                }

            window.fonetIdrar3luSecim = null;
            idrarAsamaBolumu.style.display = "none";
            labSonucBolumu.style.display = "block";

    // STANDART GİRİŞ
    async function standartGiris() {
        if (window.fonetF8Running || window.fonetDSKayitBekleniyor) return;

        window.fonetAbortRequested = false;
        window.fonetStandartGirisTamamlandi = false;
        window.fonetHastaGecmisiKapatildi = false;
        window.fonetF8Running = true;
        butonlariKilitle(true);
        durum.textContent = "Standart giriş çalışıyor...";

        try {
            // 1. KABUL
            const kabul = document.querySelector(
                'a.x-btn[role="button"][data-qtip="Kabul"]'
            );

            if (!kabul) throw new Error("Kabul butonu bulunamadı.");

            kabul.click();
            await bekle(500);

            // 2. F54 TANISI
            const tani = document.querySelector('input[name="tani"]');

            if (!tani) throw new Error("Tanı alanı bulunamadı.");

            const setter = Object.getOwnPropertyDescriptor(
                HTMLInputElement.prototype, "value"
            ).set;

            setter.call(tani, "F54");
            tani.dispatchEvent(new Event("input", { bubbles: true }));
            tani.dispatchEvent(new Event("change", { bubbles: true }));

            tani.dispatchEvent(new KeyboardEvent("keydown", {
                key: "Enter", keyCode: 13, which: 13, bubbles: true
            }));

            tani.dispatchEvent(new KeyboardEvent("keyup", {
                key: "Enter", keyCode: 13, which: 13, bubbles: true
            }));

            await bekle(500);

            // 3. DİĞER BİLGİLER
            const diger = [...document.querySelectorAll("a.x-tab")]
                .find(el => el.innerText.trim() === "Diğer Bilgiler");

            if (!diger) throw new Error("Diğer Bilgiler sekmesi bulunamadı.");

            diger.click();
            await bekle(500);

            // 4. SİGARA EVET
            const sigara = document.querySelector(
                'input[name="sigaraKullanim"]'
            );

            if (!sigara) throw new Error("Sigara alanı bulunamadı.");

            const ok = sigara.parentElement.parentElement
                .querySelector(".x-form-arrow-trigger");

            if (!ok) throw new Error("Sigara menüsü bulunamadı.");

            ok.click();
            await bekle(300);

            const evet = [...document.querySelectorAll(".x-boundlist-item")]
                .find(el => el.textContent.trim() === "EVET");

            if (!evet) throw new Error("EVET seçeneği bulunamadı.");

            evet.click();
            await bekle(500);

            // 5. HASTA GEÇMİŞİ
            const gecmis = [...document.querySelectorAll("a.x-btn")]
                .find(el =>
                    el.querySelector(".goHistory16") &&
                    el.innerText.trim() === "Hasta Geçmişi"
                );

            if (!gecmis) throw new Error("Hasta Geçmişi bulunamadı.");

            gecmis.click();
            await bekle(500);

            const gorunurKapatma = [...document.querySelectorAll("img.x-tool-img.x-tool-close")]
                .some(el => el.getClientRects().length > 0);

            if (!gorunurKapatma) {
                throw new Error("Hasta Geçmişi açıldıktan sonra görünür kapatma simgesi doğrulanamadı.");
            }

            window.fonetStandartGirisTamamlandi = true;
            durum.textContent = "Standart giriş tamamlandı. İdrar senaryosunu seçebilirsin.";
            console.log("Standart giriş otomasyonu tamamlandı; idrar senaryosu seçimine hazır.");

        } catch (err) {
            window.fonetStandartGirisTamamlandi = false;
            if (abortHatasiMi(err)) {
                durum.textContent = "ABORT MISSION: Standart Giriş durduruldu.";
                console.warn("[DS TURBO] Standart Giriş kullanıcı tarafından durduruldu.");
            } else {
                durum.textContent = "Hata oluştu.";
                alert("Standart giriş durdu: " + err.message);
                console.error(err);
            }

        } finally {
            window.fonetF8Running = false;
            butonlariKilitle(false);
        }
    }

                characterData: true

                            const satir = input.closest('.x-form-item');
                            const etiket = satir?.querySelector('.x-form-item-label');
                            const simge = satir?.querySelector('div.x-form-date-trigger[role="button"]');
                            return { input, simge, etiket: etiket?.textContent.trim() };
                        })
                    ).filter(h => h.etiket === 'DS Tarih:' &&
                        h.input.getClientRects().length > 0 &&
                        h.simge?.getClientRects().length > 0);

                    if (hedefler.length > 1) {
                        throw new Error('Birden fazla DS Tarih takvim simgesi bulundu; işlem durduruldu.');
                    }
                    if (hedefler.length === 1) tarihHedefi = hedefler[0];
                    else await bekle(200);
                }

                if (!tarihHedefi) {
                    throw new Error('DS Hasta Kartı içindeki DS Tarih simgesi 7 saniye içinde bulunamadı.');
                }

                const { input, simge } = tarihHedefi;
                const alanId = input.id.endsWith('-inputEl')
                    ? input.id.slice(0, -'-inputEl'.length) : null;
                const tarihBileseni = alanId && window.Ext?.getCmp
                    ? window.Ext.getCmp(alanId) : null;

                const takvimGorunuyorMu = () => {
                    const picker = tarihBileseni?.picker;
                    if (tarihBileseni && (tarihBileseni.isExpanded === true ||
                        (picker && typeof picker.isVisible === 'function' && picker.isVisible()))) {
                        return true;
                    }
                    // ExtJS bileşeni erişilemezse yalnızca görünür datepicker varlığını kontrol et.
                    if (tarihBileseni) return false;
                    return [...document.querySelectorAll('.x-datepicker')].some(el =>
                        el.getClientRects().length > 0);
                };

                // Render edilen alan ile olay dinleyicilerinin hazır olmasına fırsat ver.
                await bekle(450);

                if (!takvimGorunuyorMu()) {
                    simge.click();
                    await bekle(350);
                }

                if (!takvimGorunuyorMu() &&
                    tarihBileseni && typeof tarihBileseni.expand === 'function') {
                    console.warn('[DS TURBO] DS Tarih simgesine tıklama takvimi açmadı; aynı alan expand() ile deneniyor.');
                    tarihBileseni.expand();
                    await bekle(250);
                }

                if (!takvimGorunuyorMu()) {
                    throw new Error('DS Tarih simgesi bulundu fakat takvimin açıldığı doğrulanamadı; elle kontrol et.');
                }

                console.log('[DS TURBO V29] DS Tarih takvimi açıldığı doğrulandı.', {
                    input: input.outerHTML,
                    simge: simge.outerHTML,
                    extBileseniBulundu: !!tarihBileseni
                });
            }

            await bekle(300);

            console.log("DS Hasta Kartı açıldı. Manuel giriş bekleniyor.");

            // 17. DS KAYIT BAŞARISINI BEKLE
            // Eski kayıt başarı mesajı ekranda kaldıysa önce kaybolmasını,
            // ardından DS kaydına ait yeni başarı mesajını bekle.
            window.fonetDSKayitBekleniyor = true;
            butonlariKilitle(true);
            durum.textContent = islemAdi + " - DS kaydı bekleniyor...";

            const basariMesaji = "Kayıt işlemi başarılı";
            let dsMesajTemizlendi =
                !(document.body.innerText || "").includes(basariMesaji);

            if (window.fonetDSObserver) {
                window.fonetDSObserver.disconnect();
            }

            window.fonetDSObserver = new MutationObserver(() => {
                if (window.fonetAbortRequested) return;
                const sayfaMetni = document.body.innerText || "";
                const mesajVar = sayfaMetni.includes(basariMesaji);

                if (!dsMesajTemizlendi) {
                    if (!mesajVar) {
                        dsMesajTemizlendi = true;
                        console.log("Önceki başarı mesajı temizlendi; DS kayıt mesajı bekleniyor.");
                    }
                    return;
                }

                if (mesajVar) {
                    // Aynı başarı mesajının tekrar tetiklenmesini önle.
                    window.fonetDSObserver.disconnect();
                    window.fonetF8Running = true;
                    butonlariKilitle(true);
                    console.log("Yeni DS kayıt başarı mesajı yakalandı.");
                    dsKayitSonrasiIslemleri();
                }
            });

            window.fonetDSObserver.observe(document.body, {
                childList: true,
                subtree: true,
                characterData: true
            });
            abortKontrol();

            alert(
                "DS Hasta Kartı açıldı. Gerekli alanları manuel doldurup Kaydet'e bas. " +
                "Kayıt başarılı olduğunda pencere otomatik kapanacak ve Hizmet Listesi'ne DNT1 gönderilecek."
            );

        } catch (err) {
            if (abortHatasiMi(err)) {
                durum.textContent = "ABORT MISSION: " + islemAdi + " durduruldu.";
                console.warn("[DS TURBO] " + islemAdi + " kullanıcı tarafından durduruldu.");
            } else {
                durum.textContent = islemAdi + " hatası.";
                alert(islemAdi + " otomasyonu durdu: " + err.message);
                console.error(err);
            }

        } finally {
            // İptal tam DS izleyicisi kurulurken geldiyse bekleyen observer ve kilidi temizle.
            if (window.fonetAbortRequested && window.fonetDSKayitBekleniyor) {
                if (window.fonetDSObserver) {
                    window.fonetDSObserver.disconnect();
                    window.fonetDSObserver = null;
                }
                window.fonetDSKayitBekleniyor = false;
            }

            window.fonetF8Running = false;
            // DS kaydı manuel olarak bekleniyorsa tüm butonlar kilitli kalır.
            if (!window.fonetDSKayitBekleniyor) {
                butonlariKilitle(false);
            }
        }
    }

    // BUTONLARA FONKSİYONLARI BAĞLA
